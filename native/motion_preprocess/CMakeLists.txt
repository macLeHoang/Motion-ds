cmake_minimum_required(VERSION 3.18)
project(vms_motion_preprocess LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(VMS_WITH_DEEPSTREAM "Build CUDA/DeepStream library and preview" OFF)
set(DEEPSTREAM_ROOT "/opt/nvidia/deepstream/deepstream" CACHE PATH "DeepStream SDK root")

add_library(vms_geometry STATIC src/geometry.cpp)
target_include_directories(vms_geometry PUBLIC include)
target_compile_options(vms_geometry PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
  add_executable(vms_geometry_test tests/geometry_test.cpp)
  target_link_libraries(vms_geometry_test PRIVATE vms_geometry)
  add_test(NAME geometry COMMAND vms_geometry_test)
  add_executable(vms_geometry_oracle tests/geometry_oracle.cpp)
  target_link_libraries(vms_geometry_oracle PRIVATE vms_geometry)
endif()

find_package(OpenCV QUIET COMPONENTS core imgproc imgcodecs)
if(OpenCV_FOUND)
  add_library(vms_motion_core STATIC src/motion.cpp src/processor.cpp src/regions.cpp)
  target_include_directories(vms_motion_core PUBLIC include ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(vms_motion_core PUBLIC vms_geometry ${OpenCV_LIBS})
  target_compile_options(vms_motion_core PRIVATE -Wall -Wextra -Wpedantic)
  if(BUILD_TESTING)
    add_executable(vms_motion_test tests/motion_test.cpp)
    target_link_libraries(vms_motion_test PRIVATE vms_motion_core)
    add_test(NAME motion COMMAND vms_motion_test)
    add_executable(vms_regions_test tests/regions_test.cpp)
    target_link_libraries(vms_regions_test PRIVATE vms_motion_core)
    add_test(NAME regions COMMAND vms_regions_test)
    add_executable(vms_motion_oracle tests/motion_oracle.cpp)
    target_link_libraries(vms_motion_oracle PRIVATE vms_motion_core)
  endif()
elseif(VMS_WITH_DEEPSTREAM)
  message(FATAL_ERROR "OpenCV development files required (install libopencv-dev)")
else()
  message(STATUS "OpenCV development files missing: building geometry tests only")
endif()

if(BUILD_TESTING)
  find_package(Python3 QUIET COMPONENTS Interpreter)
  if(Python3_FOUND)
    add_test(NAME python_reference COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/reference_test.py --build-dir ${CMAKE_CURRENT_BINARY_DIR})
  endif()
endif()

if(VMS_WITH_DEEPSTREAM)
  enable_language(CUDA)
  set(CMAKE_CUDA_STANDARD 17)
  set(CMAKE_CUDA_STANDARD_REQUIRED ON)
  find_package(CUDAToolkit REQUIRED)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(GST REQUIRED IMPORTED_TARGET gstreamer-1.0)
  find_package(nlohmann_json 3.2.0 REQUIRED)
  set(DS_INCLUDE ${DEEPSTREAM_ROOT}/sources/includes)
  if(NOT EXISTS "${DS_INCLUDE}/nvdspreprocess_interface.h")
    message(FATAL_ERROR "DeepStream headers missing in ${DS_INCLUDE}")
  endif()
  # nvdsgst_helper holds gst_nvevent_parse_* (the preview's stream events) in DeepStream 8.0
  foreach(lib nvds_meta nvdsgst_meta nvbufsurface nvdsgst_helper)
    find_library(DS_${lib} NAMES ${lib} PATHS ${DEEPSTREAM_ROOT}/lib REQUIRED NO_DEFAULT_PATH)
  endforeach()
  add_library(vms_motion_preprocess SHARED src/preprocess.cpp src/metadata.cpp src/decoder.cpp src/gpu.cu)
  target_include_directories(vms_motion_preprocess PUBLIC include ${DS_INCLUDE})
  target_link_libraries(vms_motion_preprocess PUBLIC vms_motion_core PkgConfig::GST
    CUDA::cudart CUDA::cuda_driver nlohmann_json::nlohmann_json
    ${DS_nvds_meta} ${DS_nvdsgst_meta} ${DS_nvbufsurface} ${CMAKE_DL_LIBS})
  set_target_properties(vms_motion_preprocess PROPERTIES BUILD_RPATH "${DEEPSTREAM_ROOT}/lib")
  add_executable(vms_motion_preview src/preview.cpp)
  target_link_libraries(vms_motion_preview PRIVATE vms_motion_preprocess ${DS_nvdsgst_helper})
  target_link_libraries(vms_motion_preview PRIVATE ${CMAKE_DL_LIBS})
  set_target_properties(vms_motion_preview PROPERTIES BUILD_RPATH "${DEEPSTREAM_ROOT}/lib")
  if(BUILD_TESTING)
    add_library(vms_identity_parser SHARED src/identity_parser.cpp)
    # nvdsinfer_custom_impl.h pulls TensorRT headers that include cuda_runtime_api.h
    target_include_directories(vms_identity_parser PRIVATE ${DS_INCLUDE} ${CUDAToolkit_INCLUDE_DIRS})
    add_executable(vms_gpu_test tests/gpu_test.cpp)
    target_link_libraries(vms_gpu_test PRIVATE vms_motion_preprocess)
    add_test(NAME gpu COMMAND vms_gpu_test)
  endif()
  install(TARGETS vms_motion_preprocess vms_motion_preview LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
  install(DIRECTORY include/vms_motion DESTINATION include)
endif()
